FUNCTION findMaxInMin(arr, n)
    SET minList ← empty list

    FOR i ← 0 TO length(arr) - n DO
        SET minValue ← arr[i]
        FOR j ← i + 1 TO i + n - 1 DO
            IF arr[j] < minValue THEN
                minValue ← arr[j]
            END IF
        END FOR
        APPEND minValue TO minList
    END FOR

    SET result ← minList[0]
    FOR each value IN minList DO
        IF value > result THEN
            result ← value
        END IF
    END FOR

    RETURN result
END FUNCTION
